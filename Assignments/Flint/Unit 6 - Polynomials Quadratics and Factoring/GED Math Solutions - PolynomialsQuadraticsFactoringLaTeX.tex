\section*{SECTION 1: POLYNOMIAL OPERATIONS}

\noindent
\textbf{Question 1:} Simplify: $(3x^2 + 5x - 2) + (2x^2 - 3x + 4)$

\textbf{Solution:} Combine like terms: $3x^2 + 2x^2 = 5x^2$; $5x - 3x = 2x$; $-2 + 4 = 2$

\textbf{Answer:} $\boxed{5x^2 + 2x + 2}$

\vspace{0.3cm}
\noindent
\textbf{Question 2:} Simplify: $(7y^2 + 4y - 1) - (3y^2 + 2y - 5)$

\textbf{Solution:} Distribute the negative sign: $(7y^2 + 4y - 1) + (-3y^2 - 2y + 5)$. Combine like terms: $7y^2 - 3y^2 = 4y^2$; $4y - 2y = 2y$; $-1 + 5 = 4$

\textbf{Answer:} $\boxed{4y^2 + 2y + 4}$

\vspace{0.3cm}
\noindent
\textbf{Question 3:} Simplify: $(4a^3 + 2a^2 - a) + (a^3 - 3a^2 + 2a)$

\textbf{Solution:} Combine like terms: $4a^3 + a^3 = 5a^3$; $2a^2 - 3a^2 = -a^2$; $-a + 2a = a$

\textbf{Answer:} $\boxed{5a^3 - a^2 + a}$

\vspace{0.3cm}
\noindent
\textbf{Question 4:} Simplify: $(6b^2 - 5b + 3) - (2b^2 + b - 7)$

\textbf{Solution:} Distribute the negative sign: $(6b^2 - 5b + 3) + (-2b^2 - b + 7)$. Combine like terms: $6b^2 - 2b^2 = 4b^2$; $-5b - b = -6b$; $3 + 7 = 10$

\textbf{Answer:} $\boxed{4b^2 - 6b + 10}$

\vspace{0.3cm}
\noindent
\textbf{Question 5:} Combine like terms: $5x^2 + 3x - 2 + 4x^2 - x + 6$

\textbf{Solution:} Group like terms: $(5x^2 + 4x^2) + (3x - x) + (-2 + 6) = 9x^2 + 2x + 4$

\textbf{Answer:} $\boxed{9x^2 + 2x + 4}$

\vspace{0.3cm}
\noindent
\textbf{Question 6:} Multiply: $3x(2x^2 + 4x - 5)$

\textbf{Solution:} Distribute $3x$ to each term: $3x \cdot 2x^2 = 6x^3$; $3x \cdot 4x = 12x^2$; $3x \cdot (-5) = -15x$

\textbf{Answer:} $\boxed{6x^3 + 12x^2 - 15x}$

\vspace{0.3cm}
\noindent
\textbf{Question 7:} Multiply: $-2a(3a^2 - 5a + 1)$

\textbf{Solution:} Distribute $-2a$ to each term: $-2a \cdot 3a^2 = -6a^3$; $-2a \cdot (-5a) = 10a^2$; $-2a \cdot 1 = -2a$

\textbf{Answer:} $\boxed{-6a^3 + 10a^2 - 2a}$

\vspace{0.3cm}
\noindent
\textbf{Question 8:} Multiply: $4xy(2x^2 + 3xy - y^2)$

\textbf{Solution:} Distribute $4xy$ to each term: $4xy \cdot 2x^2 = 8x^3y$; $4xy \cdot 3xy = 12x^2y^2$; $4xy \cdot (-y^2) = -4xy^3$

\textbf{Answer:} $\boxed{8x^3y + 12x^2y^2 - 4xy^3}$

\vspace{0.3cm}
\noindent
\textbf{Question 9:} Multiply: $5m^2(m^3 + 2m - 3)$

\textbf{Solution:} Distribute $5m^2$ to each term: $5m^2 \cdot m^3 = 5m^5$; $5m^2 \cdot 2m = 10m^3$; $5m^2 \cdot (-3) = -15m^2$

\textbf{Answer:} $\boxed{5m^5 + 10m^3 - 15m^2}$

\vspace{0.3cm}
\noindent
\textbf{Question 10:} Multiply: $(x + 3)(x + 8)$

\textbf{Solution:} Using FOIL: First: $x \cdot x = x^2$; Outer: $x \cdot 8 = 8x$; Inner: $3 \cdot x = 3x$; Last: $3 \cdot 8 = 24$. Combine: $x^2 + 8x + 3x + 24 = x^2 + 11x + 24$

\textbf{Answer:} $\boxed{x^2 + 11x + 24}$

\vspace{0.3cm}
\noindent
\textbf{Question 11:} Multiply: $(2y - 1)(y - 6)$

\textbf{Solution:} Using FOIL: First: $2y \cdot y = 2y^2$; Outer: $2y \cdot (-6) = -12y$; Inner: $(-1) \cdot y = -y$; Last: $(-1) \cdot (-6) = 6$. Combine: $2y^2 - 12y - y + 6 = 2y^2 - 13y + 6$

\textbf{Answer:} $\boxed{2y^2 - 13y + 6}$

\vspace{0.3cm}
\noindent
\textbf{Question 12:} Multiply: $(a + 4)(a - 7)$

\textbf{Solution:} Using FOIL: First: $a \cdot a = a^2$; Outer: $a \cdot (-7) = -7a$; Inner: $4 \cdot a = 4a$; Last: $4 \cdot (-7) = -28$. Combine: $a^2 - 7a + 4a - 28 = a^2 - 3a - 28$

\textbf{Answer:} $\boxed{a^2 - 3a - 28}$

\section*{SECTION 2: FACTORING}

\noindent
\textbf{Question 13:} Factor: $6x^2 + 18x + 12$

\textbf{Solution:} Find the GCF of coefficients: $\gcd(6, 18, 12) = 6$. Factor out 6: $6(x^2 + 3x + 2)$. Factor the trinomial: $(x + 1)(x + 2)$

\textbf{Answer:} $\boxed{6(x + 1)(x + 2)}$

\vspace{0.3cm}
\noindent
\textbf{Question 14:} Factor: $24x^4 + 48x^3 + 56x^2$

\textbf{Solution:} Find the GCF: $\gcd(24, 48, 56) = 8$; GCF of variables: $x^2$. So GCF $= 8x^2$. Factor out: $8x^2(3x^2 + 6x + 7)$. The trinomial cannot be factored further.

\textbf{Answer:} $\boxed{8x^2(3x^2 + 6x + 7)}$

\vspace{0.3cm}
\noindent
\textbf{Question 15:} Factor: $15a^3b + 25a^2b^2 - 10ab$

\textbf{Solution:} Find the GCF: $\gcd(15, 25, 10) = 5$; GCF of variables: $ab$. So GCF $= 5ab$. Factor out: $5ab(3a^2 + 5ab - 2)$

\textbf{Answer:} $\boxed{5ab(3a^2 + 5ab - 2)}$

\vspace{0.3cm}
\noindent
\textbf{Question 16:} Factor: $12m^2 - 8m + 4$

\textbf{Solution:} Find the GCF: $\gcd(12, 8, 4) = 4$. Factor out: $4(3m^2 - 2m + 1)$. The trinomial cannot be factored further (discriminant is negative).

\textbf{Answer:} $\boxed{4(3m^2 - 2m + 1)}$

\vspace{0.3cm}
\noindent
\textbf{Question 17:} Factor: $x^2 + 4x - 12$

\textbf{Solution:} Find two numbers that multiply to $-12$ and add to $4$: $6$ and $-2$. $(6)(-2) = -12$ and $6 + (-2) = 4$

\textbf{Answer:} $\boxed{(x + 6)(x - 2)}$

\vspace{0.3cm}
\noindent
\textbf{Question 18:} Factor: $x^2 + 11x + 24$

\textbf{Solution:} Find two numbers that multiply to $24$ and add to $11$: $3$ and $8$. $(3)(8) = 24$ and $3 + 8 = 11$

\textbf{Answer:} $\boxed{(x + 3)(x + 8)}$

\vspace{0.3cm}
\noindent
\textbf{Question 19:} Factor: $y^2 - 7y + 12$

\textbf{Solution:} Find two numbers that multiply to $12$ and add to $-7$: $-3$ and $-4$. $(-3)(-4) = 12$ and $-3 + (-4) = -7$

\textbf{Answer:} $\boxed{(y - 3)(y - 4)}$

\vspace{0.3cm}
\noindent
\textbf{Question 20:} Factor: $z^2 - 2z - 15$

\textbf{Solution:} Find two numbers that multiply to $-15$ and add to $-2$: $3$ and $-5$. $(3)(-5) = -15$ and $3 + (-5) = -2$

\textbf{Answer:} $\boxed{(z + 3)(z - 5)}$

\vspace{0.3cm}
\noindent
\textbf{Question 21:} Factor: $2x^2 + 9x + 4$

\textbf{Solution:} Use trial and error. Factors of $2x^2$: $(2x)(x)$. Factors of $4$: $(1)(4)$ or $(2)(2)$. Test: $(2x + 1)(x + 4) = 2x^2 + 8x + x + 4 = 2x^2 + 9x + 4$ \checkmark

\textbf{Answer:} $\boxed{(2x + 1)(x + 4)}$

\vspace{0.3cm}
\noindent
\textbf{Question 22:} Factor: $3y^2 + 10y + 3$

\textbf{Solution:} Factors of $3y^2$: $(3y)(y)$. Factors of $3$: $(1)(3)$. Test: $(3y + 1)(y + 3) = 3y^2 + 9y + y + 3 = 3y^2 + 10y + 3$ \checkmark

\textbf{Answer:} $\boxed{(3y + 1)(y + 3)}$

\vspace{0.3cm}
\noindent
\textbf{Question 23:} Factor: $4a^2 - 11a + 6$

\textbf{Solution:} Factors of $4a^2$: $(4a)(a)$ or $(2a)(2a)$. Factors of $6$: $(1)(6)$, $(2)(3)$. Test: $(4a - 3)(a - 2) = 4a^2 - 8a - 3a + 6 = 4a^2 - 11a + 6$ \checkmark

\textbf{Answer:} $\boxed{(4a - 3)(a - 2)}$

\vspace{0.3cm}
\noindent
\textbf{Question 24:} Factor: $x^2 - 49$ (Difference of Squares)

\textbf{Solution:} Recognize as $a^2 - b^2 = (a + b)(a - b)$ where $a = x$ and $b = 7$. $x^2 - 49 = x^2 - 7^2 = (x + 7)(x - 7)$

\textbf{Answer:} $\boxed{(x + 7)(x - 7)}$

\vspace{0.3cm}
\noindent
\textbf{Question 25:} Factor: $x^2 + 6x + 9$ (Perfect Square Trinomial)

\textbf{Solution:} Recognize as $a^2 + 2ab + b^2 = (a + b)^2$ where $a = x$ and $b = 3$. Check: $x^2 + 2(x)(3) + 3^2 = x^2 + 6x + 9$ \checkmark

\textbf{Answer:} $\boxed{(x + 3)^2}$

\vspace{0.3cm}
\noindent
\textbf{Question 26:} Factor: $4a^2 - 20a + 25$ (Perfect Square Trinomial)

\textbf{Solution:} Recognize as $a^2 - 2ab + b^2 = (a - b)^2$ where $a = 2a$ and $b = 5$. Check: $(2a)^2 - 2(2a)(5) + 5^2 = 4a^2 - 20a + 25$ \checkmark

\textbf{Answer:} $\boxed{(2a - 5)^2}$

\section*{SECTION 3: QUADRATIC EQUATIONS}

\noindent
\textbf{Question 27:} Solve: $x^2 - 2x = 15$

\textbf{Solution:} Rearrange: $x^2 - 2x - 15 = 0$. Factor: $(x - 5)(x + 3) = 0$. Set each factor to zero: $x - 5 = 0$ or $x + 3 = 0$

\textbf{Answer:} $\boxed{x = 5 \text{ or } x = -3}$

\vspace{0.3cm}
\noindent
\textbf{Question 28:} Solve: $m^2 - 8m + 16 = 0$

\textbf{Solution:} Factor: $(m - 4)(m - 4) = 0$ or $(m - 4)^2 = 0$. Set to zero: $m - 4 = 0$

\textbf{Answer:} $\boxed{m = 4 \text{ (double root)}}$

\vspace{0.3cm}
\noindent
\textbf{Question 29:} Solve: $y^2 + 5y - 24 = 0$

\textbf{Solution:} Factor: Find two numbers that multiply to $-24$ and add to $5$: $8$ and $-3$. $(y + 8)(y - 3) = 0$. Set each to zero: $y + 8 = 0$ or $y - 3 = 0$

\textbf{Answer:} $\boxed{y = -8 \text{ or } y = 3}$

\vspace{0.3cm}
\noindent
\textbf{Question 30:} Solve: $2x^2 - 7x + 3 = 0$

\textbf{Solution:} Factor: $(2x - 1)(x - 3) = 0$. Set each to zero: $2x - 1 = 0$ or $x - 3 = 0$. Solve: $x = \frac{1}{2}$ or $x = 3$

\textbf{Answer:} $\boxed{x = \frac{1}{2} \text{ or } x = 3}$

\vspace{0.3cm}
\noindent
\textbf{Question 31:} Solve: $x^2 - 12x + 20 = 7$

\textbf{Solution:} Rearrange: $x^2 - 12x + 13 = 0$. Complete the square: $x^2 - 12x + 36 = -13 + 36 = 23$. $(x - 6)^2 = 23$. Take square root: $x - 6 = \pm\sqrt{23}$. Solve: $x = 6 \pm \sqrt{23}$

\textbf{Answer:} $\boxed{x = 6 + \sqrt{23} \text{ or } x = 6 - \sqrt{23}}$ (approximately $10.796$ and $1.204$)

\vspace{0.3cm}
\noindent
\textbf{Question 32:} Solve: $x^2 + 6x - 16 = 0$

\textbf{Solution:} Complete the square: $x^2 + 6x + 9 = 16 + 9 = 25$. $(x + 3)^2 = 25$. Take square root: $x + 3 = \pm 5$. Solve: $x = -3 + 5 = 2$ or $x = -3 - 5 = -8$

\textbf{Answer:} $\boxed{x = 2 \text{ or } x = -8}$

\vspace{0.3cm}
\noindent
\textbf{Question 33:} Solve: $a^2 - 8a + 12 = 0$

\textbf{Solution:} Complete the square: $a^2 - 8a + 16 = -12 + 16 = 4$. $(a - 4)^2 = 4$. Take square root: $a - 4 = \pm 2$. Solve: $a = 4 + 2 = 6$ or $a = 4 - 2 = 2$

\textbf{Answer:} $\boxed{a = 6 \text{ or } a = 2}$

\vspace{0.3cm}
\noindent
\textbf{Question 34:} Solve: $3x^2 + 11x - 4 = 0$

\textbf{Solution:} Use quadratic formula: $x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}$ where $a = 3$, $b = 11$, $c = -4$.
\[x = \frac{-11 \pm \sqrt{121 + 48}}{6} = \frac{-11 \pm \sqrt{169}}{6} = \frac{-11 \pm 13}{6}\]
Solve: $x = \frac{-11 + 13}{6} = \frac{2}{6} = \frac{1}{3}$ or $x = \frac{-11 - 13}{6} = \frac{-24}{6} = -4$

\textbf{Answer:} $\boxed{x = \frac{1}{3} \text{ or } x = -4}$

\vspace{0.3cm}
\noindent
\textbf{Question 35:} Solve: $m^2 - 4m - 6 = 0$

\textbf{Solution:} Use quadratic formula: $a = 1$, $b = -4$, $c = -6$.
\[m = \frac{4 \pm \sqrt{16 + 24}}{2} = \frac{4 \pm \sqrt{40}}{2} = \frac{4 \pm 2\sqrt{10}}{2} = 2 \pm \sqrt{10}\]

\textbf{Answer:} $\boxed{m = 2 + \sqrt{10} \text{ or } m = 2 - \sqrt{10}}$ (approximately $5.162$ and $-1.162$)

\vspace{0.3cm}
\noindent
\textbf{Question 36:} Solve: $x^2 + 5x + 2 = 0$

\textbf{Solution:} Use quadratic formula: $a = 1$, $b = 5$, $c = 2$.
\[x = \frac{-5 \pm \sqrt{25 - 8}}{2} = \frac{-5 \pm \sqrt{17}}{2}\]

\textbf{Answer:} $\boxed{x = \frac{-5 + \sqrt{17}}{2} \text{ or } x = \frac{-5 - \sqrt{17}}{2}}$ (approximately $-0.438$ and $-4.562$)

\vspace{0.3cm}
\noindent
\textbf{Question 37:} For the equation $y = x^2 - 4$, identify the vertex and the roots (x-intercepts).

\textbf{Solution:} This is a parabola in the form $y = x^2 + c$ with vertex at $(0, -4)$. For roots, set $y = 0$: $x^2 - 4 = 0$, so $x^2 = 4$, giving $x = \pm 2$

\textbf{Answer:} $\boxed{\text{Vertex: } (0, -4); \text{ Roots: } x = 2 \text{ and } x = -2}$

\vspace{0.3cm}
\noindent
\textbf{Question 38:} For the equation $y = -x^2 + 6x - 5$, determine whether the parabola opens upward or downward.

\textbf{Solution:} The coefficient of $x^2$ is $-1$, which is negative. When the coefficient of the squared term is negative, the parabola opens downward.

\textbf{Answer:} $\boxed{\text{The parabola opens downward.}}$

\vspace{0.3cm}
\noindent
\textbf{Question 39:} A rectangular garden has dimensions $(x + 3)$ meters by $(x - 2)$ meters. Write a polynomial expression for the area, then expand it.

\textbf{Solution:} Area = length $\times$ width = $(x + 3)(x - 2)$. Using FOIL: First: $x \cdot x = x^2$; Outer: $x \cdot (-2) = -2x$; Inner: $3 \cdot x = 3x$; Last: $3 \cdot (-2) = -6$. Combine: $x^2 - 2x + 3x - 6 = x^2 + x - 6$

\textbf{Answer:} $\boxed{\text{Area} = (x + 3)(x - 2) = x^2 + x - 6 \text{ square meters}}$

\vspace{0.3cm}
\noindent
\textbf{Question 40:} The height of a ball thrown upward is given by $h = -16t^2 + 64t + 5$. At what time(s) will the ball reach a height of 53 feet?

\textbf{Solution:} Set $h = 53$: $-16t^2 + 64t + 5 = 53$. Rearrange: $-16t^2 + 64t - 48 = 0$. Divide by $-16$: $t^2 - 4t + 3 = 0$. Factor: $(t - 1)(t - 3) = 0$. Solve: $t = 1$ or $t = 3$

\textbf{Answer:} $\boxed{\text{The ball reaches 53 feet at } t = 1 \text{ second and } t = 3 \text{ seconds.}}$
