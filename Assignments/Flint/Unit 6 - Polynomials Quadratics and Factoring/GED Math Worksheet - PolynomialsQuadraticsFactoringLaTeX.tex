\section*{SECTION 1: POLYNOMIAL OPERATIONS (Questions 1-12)}

\subsection*{Adding and Subtracting Polynomials}

\begin{enumerate}
\item Simplify: $(3x^2 + 5x - 2) + (2x^2 - 3x + 4)$

\item Simplify: $(7y^2 + 4y - 1) - (3y^2 + 2y - 5)$

\item Simplify: $(4a^3 + 2a^2 - a) + (a^3 - 3a^2 + 2a)$

\item Simplify: $(6b^2 - 5b + 3) - (2b^2 + b - 7)$

\item Combine like terms: $5x^2 + 3x - 2 + 4x^2 - x + 6$
\end{enumerate}

\subsection*{Multiplying by a Monomial}

\begin{enumerate}
\setcounter{enumi}{5}
\item Multiply: $3x(2x^2 + 4x - 5)$

\item Multiply: $-2a(3a^2 - 5a + 1)$

\item Multiply: $4xy(2x^2 + 3xy - y^2)$

\item Multiply: $5m^2(m^3 + 2m - 3)$
\end{enumerate}

\subsection*{Multiplying Binomials (FOIL)}

\begin{enumerate}
\setcounter{enumi}{9}
\item Multiply: $(x + 3)(x + 8)$

\item Multiply: $(2y - 1)(y - 6)$

\item Multiply: $(a + 4)(a - 7)$
\end{enumerate}
