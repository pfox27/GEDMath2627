\section*{Practice Questions}

\begin{enumerate}
    \item Classify the number $-5$ as natural, whole, integer, rational, or irrational.
    \item Write $36$ as a product of its prime factors.
    \item Find the GCF of 24 and 60.
    \item Find the LCM of 6 and 8.
    \item Simplify $\frac{45}{60}$.
    \item Convert $0.625$ to a fraction.
    \item Convert $\frac{5}{8}$ to a decimal.
    \item Add $\frac{2}{3} + \frac{4}{9}$.
    \item Subtract $\frac{5}{6} - \frac{1}{4}$.
    \item Multiply $\frac{3}{5} \times \frac{10}{9}$.
    \item Divide $\frac{7}{8} \div \frac{14}{16}$.
    \item Solve for $x$: $\frac{4}{5} = \frac{x}{25}$.
    \item Find 20\% of 250.
    \item Increase 80 by 15\%.
    \item Decrease 200 by 25\%.
    \item Round 57.438 to the nearest tenth.
    \item Estimate the sum: $198 + 402$.
    \item Evaluate $5 + 3 \times 4$.
    \item Evaluate $(6 + 2)^2 \div 4$.
    \item Find the absolute value of $-12$.
    \item Compare: Which is greater, $|-7|$ or $5$?
    \item If $4x = 28$, find $x$.
    \item A train travels 60 miles in 1.5 hours. Find its average speed.
    \item A bag has 3 red, 2 blue, and 5 green marbles. What fraction are green?
    \item Write $15\%$ as a fraction in simplest form.
    \item Simplify $3^2 + 4^2$.
    \item If 8 workers complete a task in 6 days, how many days will 12 workers take at the same rate?
    \item Convert 0.2 to a percent.
    \item The temperature drops from 12°C to -5°C. What is the total change in temperature?
    \item A car travels 240 miles in 4 hours. How far in 7 hours at the same speed?
\end{enumerate}

\section*{Solutions}

\begin{enumerate}
    \item $-5$ is an integer and rational.
    \item $36 = 2^2 \times 3^2$.
    \item GCF$(24,60) = 12$.
    \item LCM$(6,8) = 24$.
    \item $\frac{45}{60} = \frac{3}{4}$.
    \item $0.625 = \frac{625}{1000} = \frac{5}{8}$.
    \item $\frac{5}{8} = 0.625$.
    \item $\frac{2}{3} + \frac{4}{9} = \frac{6}{9} + \frac{4}{9} = \frac{10}{9} = 1\frac{1}{9}$.
    \item $\frac{5}{6} - \frac{1}{4} = \frac{10}{12} - \frac{3}{12} = \frac{7}{12}$.
    \item $\frac{3}{5} \times \frac{10}{9} = \frac{30}{45} = \frac{2}{3}$.
    \item $\frac{7}{8} \div \frac{14}{16} = \frac{7}{8} \times \frac{16}{14} = 1$.
    \item $4/5 = x/25 \Rightarrow x = 20$.
    \item $20\%$ of 250 = $0.2 \times 250 = 50$.
    \item $80 + 0.15(80) = 92$.
    \item $200 - 0.25(200) = 150$.
    \item $57.438 \approx 57.4$.
    \item Estimate: $200 + 400 = 600$.
    \item $5 + 3 \times 4 = 5 + 12 = 17$.
    \item $(6 + 2)^2 \div 4 = 64 \div 4 = 16$.
    \item $|-12| = 12$.
    \item $|-7| = 7$, and $7 > 5$.
    \item $x = 28 / 4 = 7$.
    \item Speed $= 60 / 1.5 = 40$ mph.
    \item Fraction green $= 5/10 = 1/2$.
    \item $15\% = 15/100 = 3/20$.
    \item $3^2 + 4^2 = 9 + 16 = 25$.
    \item $8 \times 6 = 12 \times x \Rightarrow x = 4$ days.
    \item $0.2 = 20\%$.
    \item Change = $12 - (-5) = 17°C$.
    \item Speed $= 240/4 = 60$ mph, distance in 7 hours = $60 \times 7 = 420$ miles.
\end{enumerate}
