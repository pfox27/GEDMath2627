\section*{Practice Questions}
\begin{enumerate}
  \item What is the value of the digit $7$ in $0.3075$? (State its place value and its value as a number.)
  \item Arrange in ascending order: $0.56,\;0.506,\;0.65$.
  \item Write $0.375$ as a fraction in simplest form.
  \item Compute: $12.5 + 3.75$.
  \item Compute: $0.08 \times 2.5$.
  \item Compute: $4.5 \div 0.15$.
  \item Simplify the fraction $\dfrac{36}{60}$.
  \item Find $x$: $\dfrac{5}{8}=\dfrac{x}{48}$.
  \item Compute: $\dfrac{3}{4}+\dfrac{2}{3}$ (express your answer as a mixed number).
  \item Compute: $\dfrac{7}{10}-\dfrac{3}{8}$.
  \item Compute: $\dfrac{5}{6}\times\dfrac{9}{10}$.
  \item Compute: $\dfrac{7}{12}\div\dfrac{14}{9}$.
  \item Convert $\dfrac{7}{20}$ to a decimal.
  \item Write $0.\overline{72}$ (repeating 72) as a fraction in simplest form.
  \item A jacket costs \$24.50. Sales tax is $6\%$. What is the total price?
  \item A recipe uses $\dfrac{3}{4}$ cup of sugar to make 12 cookies. How much sugar is needed for 18 cookies (assume direct proportion)? Give your answer as a mixed number.
  \item Find $15\%$ of $80$.
  \item Compute: $5.2-0.87+3.05$.
  \item Compute: $2\dfrac{1}{3}+1\dfrac{3}{4}$ (give a mixed number).
  \item Which is larger: $\dfrac{3}{5}$ or $0.58$? Justify your answer.
\end{enumerate}

\section*{Full Solutions}
\begin{enumerate}
  \item In $0.\underline{3}\,\underline{0}\,\underline{7}\,\underline{5}$, the $7$ is in the \textbf{thousandths} place. Its value is $7\times\tfrac{1}{1000}=\boxed{0.007=\tfrac{7}{1000}}$.
  \item Compare using place value (pad with zeros): $0.506,\;0.560,\;0.650$. Thus ascending order is $\boxed{0.506<0.56<0.65}$.
  \item $0.375=\dfrac{375}{1000}=\dfrac{375\div125}{1000\div125}=\dfrac{3}{8}$. So $\boxed{\tfrac{3}{8}}$.
  \item Align decimals: $12.50+3.75=16.25$. So $\boxed{16.25}$.
  \item $0.08\times 2.5=(8\times10^{-2})\times(25\times10^{-1})=200\times10^{-3}=\boxed{0.200=0.20}$. 
  \item $4.5\div0.15=\dfrac{45}{10}\div\dfrac{15}{100}=\dfrac{45}{10}\times\dfrac{100}{15}=\dfrac{45\times10}{15}=\dfrac{450}{15}=\boxed{30}$.
  \item $\dfrac{36}{60}=\dfrac{36\div12}{60\div12}=\dfrac{3}{5}$. So $\boxed{\tfrac{3}{5}}$.
  \item Cross-multiply: $8x=5\times48\Rightarrow 8x=240\Rightarrow x=30$. So $\boxed{30}$.
  \item $\dfrac{3}{4}+\dfrac{2}{3}=\dfrac{9}{12}+\dfrac{8}{12}=\dfrac{17}{12}=1\dfrac{5}{12}$. So $\boxed{1\dfrac{5}{12}}$.
  \item LCD $=40$: $\dfrac{7}{10}=\dfrac{28}{40}$, $\dfrac{3}{8}=\dfrac{15}{40}$. Difference $=\dfrac{28}{40}-\dfrac{15}{40}=\dfrac{13}{40}$. So $\boxed{\tfrac{13}{40}}$.
  \item $\dfrac{5}{6}\times\dfrac{9}{10}=\dfrac{45}{60}=\dfrac{3}{4}$. So $\boxed{\tfrac{3}{4}}$.
  \item $\dfrac{7}{12}\div\dfrac{14}{9}=\dfrac{7}{12}\times\dfrac{9}{14}=\dfrac{7}{14}\times\dfrac{9}{12}=\dfrac{1}{2}\times\dfrac{3}{4}=\boxed{\tfrac{3}{8}}$.
  \item $\dfrac{7}{20}=\dfrac{35}{100}=0.35$. So $\boxed{0.35}$.
  \item Let $x=0.\overline{72}$. Then $100x=72.\overline{72}$. Subtract: $100x-x=72 \Rightarrow 99x=72 \Rightarrow x=\dfrac{72}{99}=\dfrac{8}{11}$. So $\boxed{\tfrac{8}{11}}$.
  \item Tax $=0.06\times 24.50=1.47$. Total $=24.50+1.47=\boxed{\$25.97}$.
  \item Scale by factor $\dfrac{18}{12}=\dfrac{3}{2}$: $\dfrac{3}{4}\times\dfrac{3}{2}=\dfrac{9}{8}=1\dfrac{1}{8}$. So $\boxed{1\dfrac{1}{8}\text{ cup}}$.
  \item $15\% = 0.15$. Then $0.15\times80=12$. So $\boxed{12}$.
  \item $5.2-0.87=4.33$; then $4.33+3.05=7.38$. So $\boxed{7.38}$.
  \item $2\dfrac{1}{3}=\dfrac{7}{3}$ and $1\dfrac{3}{4}=\dfrac{7}{4}$. Sum $=\dfrac{28}{12}+\dfrac{21}{12}=\dfrac{49}{12}=4\dfrac{1}{12}$. So $\boxed{4\dfrac{1}{12}}$.
  \item $\dfrac{3}{5}=0.6$, and $0.6>0.58$. Therefore $\boxed{\dfrac{3}{5}\text{ is larger}}$.
\end{enumerate}
