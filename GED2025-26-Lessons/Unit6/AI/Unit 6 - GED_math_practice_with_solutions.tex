\section*{Full Solutions}
\begin{enumerate}
\item $2x + 7 = 25 \Rightarrow 2x = 18 \Rightarrow x = 9$.

\item \(5(x-3) - 2 = 3x + 4\)\\
\quad $5x - 15 - 2 = 3x + 4$\\
\quad $5x - 17 = 3x + 4$\\
\quad $2x = 21 \Rightarrow x = \dfrac{21}{2} = 10.5$.

\item \(\dfrac{x}{3} - \dfrac{5}{6} = \dfrac{1}{2}\)\\
Add $\dfrac{5}{6}$ to both sides: $\dfrac{x}{3} = \dfrac{1}{2} + \dfrac{5}{6} = \dfrac{3}{6} + \dfrac{5}{6} = \dfrac{8}{6} = \dfrac{4}{3}$.\\
Multiply both sides by 3: $x = 4$.

\item \(4(2x-1) = 3(x+5)\)\\
$8x - 4 = 3x + 15$\\
$5x = 19 \Rightarrow x = \dfrac{19}{5} = 3.8$.

\item \(7 - 2x = 3x + 12\)\\
$7 - 12 = 3x + 2x$\\
$-5 = 5x \Rightarrow x = -1$.

\item Let $t$ be the number of texts. Cost: $20 + 0.10t = 41$.\\
$0.10t = 21 \Rightarrow t = 210$.\\
\textbf{Answer:} $210$ texts.

\item \(3x - 4 \le 11\)\\
$3x \le 15 \Rightarrow x \le 5$.\\
\textbf{Interval:} $(-\infty,\,5]$.

\item \(-2 \le 2x - 6 < 8\)\\
Add 6: $4 \le 2x < 14$\\
Divide by 2: $2 \le x < 7$.\\
\textbf{Interval:} $[2,\,7)$.

\item \(|x - 5| = 9\)\\
$x - 5 = 9$ or $x - 5 = -9$ $\Rightarrow x = 14$ or $x = -4$.

\item \(|2x + 1| < 7\)\\
$-7 < 2x + 1 < 7$\\
$-8 < 2x < 6$\\
$-4 < x < 3$.\\
\textbf{Interval:} $(-4,\,3)$.

\item Substitute $y=2x-3$ into $x+y=12$:\\
$x + (2x - 3) = 12 \Rightarrow 3x = 15 \Rightarrow x = 5$.\\
Then $y = 2(5) - 3 = 7$.\\
\textbf{Solution:} $(5, 7)$.

\item Add the equations to eliminate $y$:\\
$(3x + 2y) + (5x - 2y) = 16 + 4$ gives $8x = 20 \Rightarrow x = \dfrac{20}{8} = \dfrac{5}{2}$.\\
Plug into $3x + 2y = 16$: $3\cdot\dfrac{5}{2} + 2y = 16$\\
$\dfrac{15}{2} + 2y = 16 \Rightarrow 2y = 16 - \dfrac{15}{2} = \dfrac{17}{2}$\\
$y = \dfrac{17}{4} = 4.25$.\\
\textbf{Solution:} $\left(\dfrac{5}{2}, \dfrac{17}{4}\right)$.

\item Let $s$ be student tickets, $a$ be adult tickets.\\
System: $\begin{cases} s + a = 180 \\ 6s + 10a = 1520 \end{cases}$\\
From first, $s = 180 - a$. Substitute: $6(180 - a) + 10a = 1520$\\
$1080 - 6a + 10a = 1520 \Rightarrow 4a = 440 \Rightarrow a = 110$.\\
Then $s = 180 - 110 = 70$.\\
\textbf{Answer:} $70$ student, $110$ adult.

\item $f(-2) = -3(-2) + 7 = 6 + 7 = 13$.\\
$f\!\left(\dfrac{1}{3}\right) = -3\cdot\dfrac{1}{3} + 7 = -1 + 7 = 6$.

\item Not a function. The input $x=2$ maps to two different outputs (3 and 6), which violates the definition of a function (each input must have exactly one output).

\item Slope $m = \dfrac{-6 - 2}{5 - (-3)} = \dfrac{-8}{8} = -1$.\\
Use point-slope with $( -3, 2)$: $y - 2 = -1(x + 3)$.\\
$y - 2 = -x - 3 \Rightarrow y = -x - 1$.\\
\textbf{Slope-intercept form:} $y = -x - 1$.

\item Factor: $x^2 - 9x + 20 = (x - 4)(x - 5) = 0$.\\
Solutions: $x = 4$ or $x = 5$.

\item Quadratic formula with $a=2, b=3, c=-5$:\\
$x = \dfrac{-b \pm \sqrt{b^2 - 4ac}}{2a} = \dfrac{-3 \pm \sqrt{9 - 4(2)(-5)}}{4} = \dfrac{-3 \pm \sqrt{9 + 40}}{4} = \dfrac{-3 \pm \sqrt{49}}{4}$.\\
Thus $x = \dfrac{-3 + 7}{4} = 1$ or $x = \dfrac{-3 - 7}{4} = -\dfrac{10}{4} = -\dfrac{5}{2}$.

\item Complete the square or use $x = -\dfrac{b}{2a}$. Here $a=1, b=-6$.\\
Axis: $x = -\dfrac{-6}{2(1)} = \dfrac{6}{2} = 3$.\\
Vertex: Evaluate $y$ at $x=3$: $y = 3^2 - 6(3) + 11 = 9 - 18 + 11 = 2$.\\
\textbf{Axis:} $x=3$. \textbf{Vertex:} $(3, 2)$.

\item $(f\circ g)(3) = f(g(3))$.\\
$g(3) = 3^2 - 4 = 9 - 4 = 5$. Then $f(5) = 2(5) + 1 = 11$.\\
$(g\circ f)(3) = g(f(3))$. $f(3) = 2(3) + 1 = 7$. Then $g(7) = 7^2 - 4 = 49 - 4 = 45$.\\
\textbf{Answers:} $(f\circ g)(3) = 11$, $(g\circ f)(3) = 45$.

\end{enumerate}
