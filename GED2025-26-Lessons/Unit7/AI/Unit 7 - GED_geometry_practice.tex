\section*{Practice Questions}

\subsection*{1. Points, Lines, and Angles}
1. Find the complement of a 42$^\circ$ angle.\\
2. Find the supplement of a 128$^\circ$ angle.\\
3. Two complementary angles are in ratio 3:2. Find both.\\
4. Two angles form a linear pair. One is 115$^\circ$. Find the other.\\
5. If two vertical angles are $(3x+10)^\circ$ and $(5x-30)^\circ$, find $x$.

\subsection*{2. Triangles}
6. Find the missing angle in a triangle with angles 47$^\circ$ and 83$^\circ$.\\
7. A right triangle has legs 6 and 8. Find the hypotenuse.\\
8. Find the area of a triangle with base 9 and height 12.\\
9. The sides of a triangle are 5, 12, and 13. Determine if it is a right triangle.\\
10. An isosceles triangle has equal sides 10 and base 12. Find its height.

\subsection*{3. Quadrilaterals}
11. Find the area of a rectangle with $l=15$ and $w=4$.\\
12. Find the area of a parallelogram with base 14 and height 8.\\
13. A trapezoid has bases 10 and 16 and height 6. Find its area.\\
14. A square has diagonal 12. Find its side length.\\
15. A rhombus has diagonals 10 and 24. Find its area.

\subsection*{4. Circles}
16. Find the circumference of a circle with radius 9.\\
17. Find the area of a circle with diameter 10.\\
18. The circumference of a circle is 62.8. Find its radius.\\
19. A circle has area 113.04. Find its radius.\\
20. Find the area of a semicircle with diameter 8.

\newpage
\section*{Solutions}

\subsection*{1. Points, Lines, and Angles}
1. $90 - 42 = 48^\circ$.\\
2. $180 - 128 = 52^\circ$.\\
3. $3x + 2x = 90 \Rightarrow 5x = 90 \Rightarrow x = 18$. Angles: $54^\circ$ and $36^\circ$.\\
4. $180 - 115 = 65^\circ$.\\
5. $3x + 10 = 5x - 30 \Rightarrow 40 = 2x \Rightarrow x = 20$.

\subsection*{2. Triangles}
6. $180 - (47 + 83) = 50^\circ$.\\
7. $c = \sqrt{6^2 + 8^2} = 10$.\\
8. $A = \tfrac{1}{2}(9)(12) = 54$.\\
9. $5^2 + 12^2 = 13^2 \Rightarrow 25 + 144 = 169$. True, it is right.\\
10. Height $= \sqrt{10^2 - 6^2} = 8$.

\subsection*{3. Quadrilaterals}
11. $A = 15 \times 4 = 60$.\\
12. $A = 14 \times 8 = 112$.\\
13. $A = \tfrac{1}{2}(10 + 16)(6) = 78$.\\
14. $s = \tfrac{12}{\sqrt{2}} = 6\sqrt{2}$.\\
15. $A = \tfrac{1}{2}(10)(24) = 120$.

\subsection*{4. Circles}
16. $C = 2\pi r = 18\pi$.\\
17. $A = \pi r^2 = \pi (5)^2 = 25\pi$.\\
18. $r = \tfrac{62.8}{2\pi} = 10$.\\
19. $r = \sqrt{\tfrac{113.04}{\pi}} = 6$.\\
20. $A = \tfrac{1}{2}\pi r^2 = \tfrac{1}{2}\pi(4)^2 = 8\pi$.
