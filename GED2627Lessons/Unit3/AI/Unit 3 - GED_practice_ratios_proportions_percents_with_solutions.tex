\section*{Instructions}
Show your work. Calculators are allowed. Unless stated otherwise, round money to the nearest cent and other answers to the nearest tenth.

\section*{Practice Questions (20)}
\begin{enumerate}

% 1
\item Simplify the ratio $18:24$.

% 2
\item The ratio of cats to dogs at a shelter is $12$ to $15$. Write this ratio in simplest form.

% 3
\item A class has $14$ girls and $11$ boys. Write the ratio of boys to total students in simplest form.

% 4
\item Convert the ratio $150\text{ cm} : 2\text{ m}$ to simplest form.

% 5
\item A recipe uses $3$ cups of flour and $2$ cups of sugar. If you use $7.5$ cups of flour, how much sugar keeps the ratio the same?

% 6
\item Solve the proportion: $\dfrac{x}{12} = \dfrac{5}{8}$.

% 7
\item Solve the proportion: $\dfrac{14}{x} = \dfrac{7}{9}$.

% 8
\item On a map, $1$ cm represents $8$ km. How many kilometers are represented by $3.6$ cm?

% 9
\item A car travels $210$ miles on $7$ gallons of gas. At the same rate, how far can it travel on $10$ gallons?

% 10
\item A store mixes nuts in a ratio of $2$ parts almonds to $3$ parts peanuts. If the final mix weighs $10$ lb, how many pounds are almonds?

% 11
\item Find $35\%$ of $480$.

% 12
\item $54$ is what percent of $90$?

% 13
\item $27$ is $15\%$ of what number?

% 14
\item An item increases in price from $\$64$ to $\$80$. Find the percent increase.

% 15
\item A jacket marked $\$120$ is discounted $25\%$. What is the sale price?

% 16
\item A meal costs $\$48$ before tax. Sales tax is $13\%$ and you tip $15\%$ on the pre-tax amount. What is the total cost?

% 17
\item Simple interest: You invest $\$600$ at an annual simple interest rate of $4\%$ for $3$ years. How much interest is earned, and what is the final amount?

% 18
\item A population drops from $1{,}250$ to $1{,}100$. Find the percent decrease.

% 19
\item In a survey, $120$ people prefer tea and $180$ prefer coffee. What percent prefer tea?

% 20
\item Multi-step: A laptop has a list price of $\$950$. It goes on sale for $20\%$ off, then a coupon takes an additional $10\%$ off the sale price. After discounts, $13\%$ tax is added. What is the final price?

\end{enumerate}

\section*{Full Solutions}
\begin{enumerate}

% 1
\item \begin{solution}
$18:24 = \dfrac{18}{24} = \dfrac{3}{4}$ after dividing by $6$. So $3:4$.
\end{solution}

% 2
\item \begin{solution}
$12:15 = \dfrac{12}{15} = \dfrac{4}{5}$ after dividing by $3$. So $4:5$.
\end{solution}

% 3
\item \begin{solution}
Total $=14+11=25$. Boys:total $=11:25$ (already simplest).
\end{solution}

% 4
\item \begin{solution}
$2\,\text{m}=200\,\text{cm}$. So $150:200=\dfrac{150}{200}=\dfrac{3}{4}$, i.e., $3:4$.
\end{solution}

% 5
\item \begin{solution}
Flour:sugar $=3:2=\dfrac{3}{2}$. Let sugar $=x$. Set $\dfrac{3}{2}=\dfrac{7.5}{x}\Rightarrow 3x=15\Rightarrow x=5$. Use $5$ cups sugar.
\end{solution}

% 6
\item \begin{solution}
$\dfrac{x}{12}=\dfrac{5}{8}\Rightarrow 8x=60\Rightarrow x=7.5$.
\end{solution}

% 7
\item \begin{solution}
$\dfrac{14}{x}=\dfrac{7}{9}\Rightarrow 7x=126\Rightarrow x=18$.
\end{solution}

% 8
\item \begin{solution}
Scale $1\,\text{cm}:8\,\text{km}$. So $3.6\,\text{cm}\to 3.6\times 8=28.8\,\text{km}$.
\end{solution}

% 9
\item \begin{solution}
Unit rate $=210/7=30\,\text{mpg}$. For $10$ gal: $30\times10=300\,$miles.
\end{solution}

% 10
\item \begin{solution}
Total parts $=2+3=5$. Almonds $=\dfrac{2}{5}\times 10=4\,$lb.
\end{solution}

% 11
\item \begin{solution}
$0.35\times480=168$.
\end{solution}

% 12
\item \begin{solution}
$\dfrac{54}{90}=0.6=60\%$.
\end{solution}

% 13
\item \begin{solution}
$27=0.15\,N\Rightarrow N=\dfrac{27}{0.15}=180$.
\end{solution}

% 14
\item \begin{solution}
Change $=80-64=16$. Percent increase $=\dfrac{16}{64}=0.25=25\%$.
\end{solution}

% 15
\item \begin{solution}
Discount $=0.25\times120=30$. Sale price $=120-30=\$90.00$.
\end{solution}

% 16
\item \begin{solution}
Tax $=0.13\times48=\$6.24$. Tip (on pre-tax) $=0.15\times48=\$7.20$. Total $=48+6.24+7.20=\$61.44$.
\end{solution}

% 17
\item \begin{solution}
Simple interest $I=Prt=600\times0.04\times3=\$72$. Final amount $=600+72=\$672$.
\end{solution}

% 18
\item \begin{solution}
Change $=1{,}250-1{,}100=150$. Percent decrease $=\dfrac{150}{1{,}250}=0.12=12\%$.
\end{solution}

% 19
\item \begin{solution}
Total $=120+180=300$. Percent tea $=\dfrac{120}{300}=0.4=40\%$.
\end{solution}

% 20
\item \begin{solution}
First discount: $20\%$ of $950$ is $190$. Sale price $=950-190=760$. Second discount: $10\%$ of $760$ is $76$. New price $=760-76=684$. Tax: $13\%$ of $684$ is $88.92$. Final price $=684+88.92=\$772.92$.
\end{solution}

\end{enumerate}
