\textbf{Area}\\
\begin{tabular}{ |c| |c| |c| }
\hline
square: A = $s^2$ & rectangle: A = LW & triangle: A = $\frac{1}{2}bh$ \\
\hline
circle: A = $\pi r^2$ & parallelogram: A = bh & trapezoid: A = $ \left(\frac{1}{2} \right)h(b_1 + b_2)$ \\
\hline

\end{tabular}

\textbf{Surface Area and Volume}\\
\begin{tabular}{ |c| |c| |c| }
\hline
rectangular/right prism & SA = ph + 2B & V = Bh  \\ 
\hline
cylinder & SA = 2$\pi$h + 2$\pi r^2$ &  V = $\pi r^2h$ \\ 
\hline
pyramid & SA = $\left(\frac{1}{2}\right)ps$ + B & V = $\left(\frac{1}{3}\right)Bh$ \\ 
\hline
cone & SA = $\pi rs$ + $\pi r^2$ & V = $\left(\frac{1}{3}\right)\pi r^2h$ \\
\hline 
sphere & SA = 4$\pi r^2$ & V = $ \left(\frac{4}{3} \right)\pi r^3$ \\
\hline
\end{tabular}

\textbf{Algebra}\\
\begin{tabular}{ |c| |c| }
\hline
slope of a line & m = $\frac{(y_2 \: - \: y_1)}{(x_2 \: - \: x_1)}$  \\ 
\hline
equation of a line (slope-intercept) & y = mx + b  \\ 
\hline
equation of a line (point-slope) & y - $y_1$ = m(x - $x_1$) \\ 
\hline
quadratic equation (standard form) & $y \: = \: ax^2 \: + bx \: + c$ \\
\hline 
quadratic formula & x = $\frac{-b \: \pm \: \sqrt{b^2 \: - \: 4ac}}{2a}$ \\
\hline
pythagorean theorem & x = $a^2 \: + b^2 \: = c^2$ \\
\hline
simple interest & I = prt \\
\hline
distance formula & d = rt \\
\hline
total cost & number of units $\times$ price per unit \\
\hline
\end{tabular}

(I = interest, p = principal, r = rate, t = time)
